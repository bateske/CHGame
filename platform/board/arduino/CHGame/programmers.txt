# Programmers offered for Burn Bootloader and Upload Using Programmer.
#
# Normal uploads never come here. A programmer is for changing the bootloader
# itself, and there are two ways to do that. Which bootloader is written is
# the board's Tools > Bootloader menu (boards.txt).
#
# "CHGame USB" goes through the bootloader that is already on the board, the
# same way Upload does: no driver, no buttons. Every CHGame leaves the factory
# with a bootloader, so this is the one to use.
#
# "WCH factory ISP" is not an external programmer either - it is the immutable
# ISP built into the chip, entered by holding BOOT across a power cycle (and on
# Windows it needs the WinUSB driver, docs/recovery.md). It is for a board
# whose bootloader is missing, damaged or locked.
#
# burn.method and burn.port are read by the recipes in platform.txt.

wchisp.name=WCH factory ISP (Hold BOOT on power-up, requires driver)
wchisp.program.tool=chgame_upload
wchisp.program.extra_params=
wchisp.burn.method=isp
wchisp.burn.port=

chgameusb.name=CHGame USB (requires CHGame bootloader, no drivers)
chgameusb.program.tool=chgame_upload
chgameusb.program.extra_params=
chgameusb.burn.method=usb
chgameusb.burn.port=-port "{serial.port}"
